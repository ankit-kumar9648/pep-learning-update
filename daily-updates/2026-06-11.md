# C++ Loop Programs - PEP Class

## 1. Reverse the Number

```cpp
#include <iostream>
using namespace std;

int main() {
    int n, rev = 0;

    cout << "Enter a number: ";
    cin >> n;

    while (n != 0) {
        rev = rev * 10 + n % 10;
        n /= 10;
    }

    cout << "Reversed Number = " << rev;
    return 0;
}
```

## 3. Check Prime Number

```cpp
#include <iostream>
using namespace std;

int main() {
    int n;
    bool isPrime = true;

    cout << "Enter a number: ";
    cin >> n;

    if (n <= 1)
        isPrime = false;
    else {
        for (int i = 2; i * i <= n; i++) {
            if (n % i == 0) {
                isPrime = false;
                break;
            }
        }
    }

    if (isPrime)
        cout << n << " is Prime.";
    else
        cout << n << " is Not Prime.";

    return 0;
}
```

## 4. Check Armstrong Number

```cpp
#include <iostream>
using namespace std;

int main() {
    int n, temp, rem, sum = 0;

    cout << "Enter a number: ";
    cin >> n;

    temp = n;

    while (temp != 0) {
        rem = temp % 10;
        sum += rem * rem * rem;
        temp /= 10;
    }

    if (sum == n)
        cout << n << " is an Armstrong Number.";
    else
        cout << n << " is Not an Armstrong Number.";

    return 0;
}
```

## 12. Fibonacci Series

```cpp

int main() {
    int n, a = 0, b = 1, c;

    cout << "Enter number of terms: ";
    cin >> n;

    cout << a << " " << b << " ";

    for (int i = 3; i <= n; i++) {
        c = a + b;
        cout << c << " ";

        a = b;
        b = c;
    }

    return 0;
}
```

## 14. Sum of Digits Till Single Value

```cpp
#include <iostream>
using namespace std;

int main() {
    int n, sum;

    cout << "Enter a number: ";
    cin >> n;

    while (n >= 10) {
        sum = 0;

        while (n > 0) {
            sum += n % 10;
            n /= 10;
        }

        n = sum;
    }

    cout << "Single Digit Sum = " << n;

    return 0;
}
```
