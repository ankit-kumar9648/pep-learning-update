## 1. Sum of Two Numbers

```cpp
#include <iostream>
using namespace std;

int main() {
    int a, b;
    cout << "Enter two numbers: ";
    cin >> a >> b;

    cout << "Sum = " << a + b;
    return 0;
}
```

## 3. Swapping Two Numbers

```cpp
#include <iostream>
using namespace std;

int main() {
    int a, b, temp;

    cout << "Enter two numbers: ";
    cin >> a >> b;

    temp = a;
    a = b;
    b = temp;

    cout << "After swapping:" << endl;
    cout << "a = " << a << endl;
    cout << "b = " << b;

    return 0;
}
```

## 4. Convert Fahrenheit to Celsius

```cpp
#include <iostream>
using namespace std;

int main() {
    float fahrenheit, celsius;

    cout << "Enter temperature in Fahrenheit: ";
    cin >> fahrenheit;

    celsius = (fahrenheit - 32) * 5 / 9;

    cout << "Temperature in Celsius = " << celsius;

    return 0;
}
```

## 8. Greater Among Two Numbers (Simple Method)

```cpp
#include <iostream>
using namespace std;

int main() {
    int a, b;

    cout << "Enter two numbers: ";
    cin >> a >> b;

    if (a > b)
        cout << a << " is greater.";
    else
        cout << b << " is greater.";

    return 0;
}
```

## 9. Greater Among Two Numbers (Using if-else)

```cpp
#include <iostream>
using namespace std;

int main() {
    int a, b;

    cout << "Enter two numbers: ";
    cin >> a >> b;

    if (a > b) {
        cout << a << " is greater.";
    }
    else if (b > a) {
        cout << b << " is greater.";
    }
    else {
        cout << "Both numbers are equal.";
    }

    return 0;
}
```
