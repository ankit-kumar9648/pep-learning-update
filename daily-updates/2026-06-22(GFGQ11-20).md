## Q11. Check Palindrome Number

```cpp
#include <iostream>
using namespace std;

int main()
{
    int n;
    cout << "Enter number: ";
    cin >> n;

    int original = n;
    int reversed = 0;

    while(n > 0)
    {
        int digit = n % 10;
        reversed = reversed * 10 + digit;
        n /= 10;
    }

    if(original == reversed)
        cout << "Palindrome Number";
    else
        cout << "Not a Palindrome Number";

    return 0;
}
```

### Output

```text
Input: 121
Output: Palindrome Number
```

## Q12. Fibonacci Number Using Recursion

```cpp
#include <iostream>
using namespace std;

int fibonacci(int n)
{
    if(n == 0)
        return 0;

    if(n == 1)
        return 1;

    return fibonacci(n - 1) + fibonacci(n - 2);
}

int main()
{
    int n;
    cout << "Enter N: ";
    cin >> n;

    cout << fibonacci(n);

    return 0;
}
```

### Output

```text
Input: 12
Output: 144
```

## Q14. Check if Two Arrays Are Equal

```cpp
#include <iostream>
using namespace std;

int main()
{
    int n1, n2;

    cin >> n1;
    int arr1[n1];

    for(int i = 0; i < n1; i++)
        cin >> arr1[i];

    cin >> n2;
    int arr2[n2];

    for(int i = 0; i < n2; i++)
        cin >> arr2[i];

    if(n1 != n2)
    {
        cout << "Arrays are not equal";
        return 0;
    }

    bool equal = true;

    for(int i = 0; i < n1; i++)
    {
        if(arr1[i] != arr2[i])
        {
            equal = false;
            break;
        }
    }

    if(equal)
        cout << "Arrays are equal";
    else
        cout << "Arrays are not equal";

    return 0;
}
```

### Output

```text
Input:
3
5 8 3
3
5 8 3

Output:
Arrays are equal
```

## Q15. Calculate Average of Array Elements

```cpp
#include <iostream>
using namespace std;

int main()
{
    int n;

    int sum = 0;
```

## Q16. Find Maximum and Minimum in Array

```cpp
#include <iostream>
using namespace std;

int main()
{
    int n;
    cin >> n;

    int arr[n];

    for(int i = 0; i < n; i++)
        cin >> arr[i];

    int maximum = arr[0];
    int minimum = arr[0];

    for(int i = 1; i < n; i++)
    {
        if(arr[i] > maximum)
            maximum = arr[i];

        if(arr[i] < minimum)
            minimum = arr[i];
    }

    cout << "Maximum = " << maximum << endl;
    cout << "Minimum = " << minimum;

    return 0;
}
```

### Output

```text
Input:
6
10 12 45 48 22 18

Output:
Maximum = 48
Minimum = 10
```

## Q17. Linear Search in Array

```cpp
#include <iostream>
using namespace std;

int main()
{
    int n;
    cin >> n;

    int arr[n];

    for(int i = 0; i < n; i++)
        cin >> arr[i];

    int key;
    cin >> key;

    int index = -1;

    for(int i = 0; i < n; i++)
    {
        if(arr[i] == key)
        {
            index = i;
            break;
        }
    }

    if(index != -1)
        cout << key << " found at index " << index;
    else
        cout << key << " not found";

    return 0;
}
```

### Output

```text
Input:
6
10 11 7 8 2 9
8

Output:
8 found at index 3
```

## Q18. Right Rotate Array K Times

```cpp
#include <iostream>
using namespace std;

int main()
{
    int n;
    cin >> n;

    int arr[n];

    for(int i = 0; i < n; i++)
        cin >> arr[i];

    int k;
    cin >> k;

    k = k % n;

    for(int i = 0; i < k; i++)
    {
        int last = arr[n - 1];

        for(int j = n - 1; j > 0; j--)
            arr[j] = arr[j - 1];

        arr[0] = last;
    }

    for(int i = 0; i < n; i++)
        cout << arr[i] << " ";

    return 0;
}
```

### Output

```text
Input:
6
10 11 7 33 5 1
3

Output:
33 5 1 10 11 7
```

## Q19. Simple Calculator

```cpp
#include <iostream>
using namespace std;

int main()
{
    int a, b;
    char op;

    cin >> a >> b;
    cin >> op;

    switch(op)
    {
        case '+':
            cout << a + b;
            break;

        case '-':
            cout << a - b;
            break;

        case '*':
            cout << a * b;
            break;

        case '/':
            cout << a / b;
            break;

        default:
            cout << "Invalid Operator";
    }

    return 0;
}
```

## Q20. Print Prime Numbers from 1 to N

```cpp
#include <iostream>
using namespace std;

int main()
{
    int n;
    cin >> n;

    for(int num = 2; num <= n; num++)
    {
        bool isPrime = true;

        for(int i = 2; i * i <= num; i++)
        {
            if(num % i == 0)
            {
                isPrime = false;
                break;
            }
        }

        if(isPrime)
            cout << num << " ";
    }

    return 0;
}
```

### Output

```text
Input:
20

Output:
2 3 5 7 11 13 17 19
```
